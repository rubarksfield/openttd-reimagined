cmake_minimum_required(VERSION 3.24)
project(OpenTTDPad VERSION 0.1.0 LANGUAGES C CXX OBJCXX)
set(OPENTTD_DEVELOPMENT_TEAM "" CACHE STRING "Apple development team; empty produces an unsigned local build")
set(OPENTTD_BUNDLE_IDENTIFIER "org.openttd.reimagined.dev" CACHE STRING "Application identifier; use your own registered identifier for distribution")
option(OPENTTD_RUNTIME_EVIDENCE "Write local runtime evidence for prototype verification" ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 message(FATAL_ERROR "Use an iOS or iOS Simulator SDK")
endif()
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROJECT_ROOT}/native/apple/Info.plist")
set(UPSTREAM "${PROJECT_ROOT}/vendor/openttd-15.3")
set(GENERATED "${PROJECT_ROOT}/build/reference/generated")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
foreach(TOOL XZ XZDEC LZMADEC LZMAINFO SCRIPTS)
 set(XZ_TOOL_${TOOL} OFF CACHE BOOL "" FORCE)
endforeach()
set(ENABLE_NLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PROJECT_ROOT}/vendor/native-deps/xz-5.8.1" xz EXCLUDE_FROM_ALL)
include("${PROJECT_ROOT}/cmake/native-archive.cmake")
openttd_add_native_archive("${PROJECT_ROOT}")
add_library(native_content_import STATIC "${PROJECT_ROOT}/native/shared/ContentImporter.cpp")
target_compile_options(native_content_import PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(native_content_import PUBLIC "${PROJECT_ROOT}/native/shared"
 PRIVATE "${UPSTREAM}/src/3rdparty")
target_link_libraries(native_content_import PUBLIC archive_static)
include("${PROJECT_ROOT}/build/ipad-engine-sources.cmake")
add_library(openttd_engine OBJECT ${OPENTTD_ENGINE_SOURCES} "${PROJECT_ROOT}/native/apple/EngineBridge.cpp" "${PROJECT_ROOT}/native/shared/TutorialGuide.cpp")
target_compile_definitions(openttd_engine PRIVATE UNIX POINTER_IS_64BIT WITH_ASSERT WITH_ZLIB WITH_LIBLZMA)
target_compile_options(openttd_engine PRIVATE -Wno-multichar -fno-strict-enums)
target_include_directories(openttd_engine PUBLIC "${UPSTREAM}/src" "${UPSTREAM}/src/3rdparty/squirrel/include" "${GENERATED}" "${GENERATED}/script" "${GENERATED}/script/api" "${GENERATED}/script/api/ai" "${GENERATED}/script/api/game" "${GENERATED}/script/api/template" "${UPSTREAM}/src/script/api" "${UPSTREAM}/src/script/api/ai" "${UPSTREAM}/src/script/api/game" "${UPSTREAM}/src/script/api/template" "${PROJECT_ROOT}/native/apple")
target_include_directories(openttd_engine PRIVATE "${PROJECT_ROOT}/vendor/native-deps/xz-5.8.1/src/liblzma/api")
target_link_libraries(openttd_engine PUBLIC liblzma z)
target_precompile_headers(openttd_engine PRIVATE "${UPSTREAM}/src/stdafx.h" "${UPSTREAM}/src/core/format.hpp")
set_source_files_properties("${UPSTREAM}/src/3rdparty/fmt/format.cc" PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
set_source_files_properties("${UPSTREAM}/src/widget.cpp" "${UPSTREAM}/src/dropdown.cpp" "${UPSTREAM}/src/fios_gui.cpp" "${UPSTREAM}/src/network/network_content_gui.cpp" "${UPSTREAM}/src/newgrf_gui.cpp" "${UPSTREAM}/src/depot_gui.cpp" "${UPSTREAM}/src/order_gui.cpp" "${UPSTREAM}/src/window.cpp" "${UPSTREAM}/src/intro_gui.cpp" "${UPSTREAM}/src/help_gui.cpp" PROPERTIES COMPILE_DEFINITIONS OTTD_TOUCH_UI)
set_source_files_properties("${PROJECT_ROOT}/native/apple/EngineBridge.cpp" PROPERTIES COMPILE_DEFINITIONS "$<$<BOOL:${OPENTTD_RUNTIME_EVIDENCE}>:OTTD_RUNTIME_EVIDENCE>")
set_source_files_properties("${PROJECT_ROOT}/native/apple/AudioDriver.mm" PROPERTIES COMPILE_DEFINITIONS "$<$<BOOL:${OPENTTD_RUNTIME_EVIDENCE}>:OTTD_RUNTIME_EVIDENCE>")
add_executable(OpenTTDPad MACOSX_BUNDLE "${PROJECT_ROOT}/native/apple/App.mm" "${PROJECT_ROOT}/native/apple/PlatformServices.mm" "${PROJECT_ROOT}/native/apple/AudioDriver.mm"
 "${PROJECT_ROOT}/native/apple/ContentImportController.mm" "${PROJECT_ROOT}/native/apple/ContentPathKey.mm"
 "${PROJECT_ROOT}/native/apple/TutorialController.mm" "${PROJECT_ROOT}/native/apple/ManagedResources.mm"
 "${PROJECT_ROOT}/native/apple/Tutorial/first-steps.json")
set_source_files_properties("${PROJECT_ROOT}/native/apple/Tutorial/first-steps.json" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
target_sources(OpenTTDPad PRIVATE "${PROJECT_ROOT}/native/apple/PrivacyInfo.xcprivacy")
set_source_files_properties("${PROJECT_ROOT}/native/apple/PrivacyInfo.xcprivacy" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
set_source_files_properties("${UPSTREAM}/src/story_gui.cpp" PROPERTIES COMPILE_DEFINITIONS OTTD_TOUCH_UI)
if(CMAKE_GENERATOR STREQUAL "Xcode")
 set_property(GLOBAL PROPERTY XCODE_EMIT_EFFECTIVE_PLATFORM_NAME ON)
 # An archive must contain the app in Products/Applications. CMake's default
 # empty install path otherwise produces a successful but empty generic archive.
 set_target_properties(OpenTTDPad PROPERTIES XCODE_ATTRIBUTE_SKIP_INSTALL NO XCODE_ATTRIBUTE_INSTALL_PATH "$(LOCAL_APPS_DIR)")
 foreach(native_library openttd_engine native_content_import liblzma archive_static)
  set_target_properties(${native_library} PROPERTIES XCODE_ATTRIBUTE_SKIP_INSTALL YES)
 endforeach()
 # File Provider can reattach FinderInfo to .app directories in Documents,
 # invalidating Apple's signing rules. Keep signable products in local
 # DerivedData; source, object files and player documents stay in place.
 string(SHA256 native_product_key "${PROJECT_ROOT}")
 string(SUBSTRING "${native_product_key}" 0 12 native_product_key)
 set(OPENTTD_XCODE_PRODUCTS_DIR "$ENV{HOME}/Library/Developer/Xcode/DerivedData/OpenTTDReimagined-${native_product_key}/Products" CACHE PATH "Local directory for signable Xcode app products")
 set_target_properties(OpenTTDPad PROPERTIES XCODE_ATTRIBUTE_CONFIGURATION_BUILD_DIR "${OPENTTD_XCODE_PRODUCTS_DIR}/$(CONFIGURATION)$(EFFECTIVE_PLATFORM_NAME)")
 target_sources(OpenTTDPad PRIVATE "${PROJECT_ROOT}/native/apple/Assets.xcassets")
 set_source_files_properties("${PROJECT_ROOT}/native/apple/Assets.xcassets" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
 set_target_properties(OpenTTDPad PROPERTIES XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME AppIcon)
endif()
target_compile_options(OpenTTDPad PRIVATE -fobjc-arc -Wno-multichar)
target_link_libraries(OpenTTDPad PRIVATE openttd_engine native_content_import "-framework UIKit" "-framework Metal" "-framework MetalKit" "-framework Foundation" "-framework CoreGraphics" "-framework QuartzCore" "-framework AVFoundation" "-framework AudioToolbox" "-framework UniformTypeIdentifiers")
set_target_properties(OpenTTDPad PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "${OPENTTD_BUNDLE_IDENTIFIER}" MACOSX_BUNDLE_BUNDLE_NAME "OpenTTD" MACOSX_BUNDLE_INFO_PLIST "${PROJECT_ROOT}/native/apple/Info.plist" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO XCODE_ATTRIBUTE_ENABLE_USER_SCRIPT_SANDBOXING NO)
set_target_properties(OpenTTDPad PROPERTIES XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${OPENTTD_BUNDLE_IDENTIFIER}")
if(CMAKE_GENERATOR STREQUAL "Xcode" OR OPENTTD_DEVELOPMENT_TEAM)
 set_target_properties(OpenTTDPad PROPERTIES XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED YES XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic)
 if(OPENTTD_DEVELOPMENT_TEAM)
  set_target_properties(OpenTTDPad PROPERTIES XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${OPENTTD_DEVELOPMENT_TEAM}")
 endif()
endif()
if(CMAKE_GENERATOR STREQUAL "Xcode")
 add_custom_command(TARGET OpenTTDPad POST_BUILD
  COMMAND /bin/bash "${PROJECT_ROOT}/script/package_xcode_app.sh" "${PROJECT_ROOT}" "${CMAKE_COMMAND}" VERBATIM)
else()
 add_custom_command(TARGET OpenTTDPad POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_ROOT}/build/ipad-resources" "$<TARGET_BUNDLE_DIR:OpenTTDPad>/GameData"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_ROOT}/native/apple/AppIcons" "$<TARGET_BUNDLE_DIR:OpenTTDPad>" VERBATIM)
endif()
