cmake_minimum_required(VERSION 3.24)
project(NativeContentImportTests LANGUAGES C CXX OBJCXX)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(IMPORT_SANITIZERS "Enable address and undefined behavior sanitizers" ON)
if(IMPORT_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
include("${PROJECT_ROOT}/cmake/native-archive.cmake")
openttd_add_native_archive("${PROJECT_ROOT}")
add_executable(content_import_cli "${PROJECT_ROOT}/native/shared/ContentImporter.cpp"
 "${PROJECT_ROOT}/native/apple/ContentPathKey.mm" "${PROJECT_ROOT}/native/tests/content_import_cli.cpp")
target_include_directories(content_import_cli PRIVATE "${PROJECT_ROOT}/native/shared"
 "${PROJECT_ROOT}/native/apple" "${PROJECT_ROOT}/vendor/openttd-15.3/src/3rdparty")
target_link_libraries(content_import_cli PRIVATE archive_static "-framework Foundation")
target_compile_options(content_import_cli PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(content_import_cli PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
